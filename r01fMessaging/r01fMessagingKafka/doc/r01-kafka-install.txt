 [2.] Crear Script Arranque : kafka-dena.bat : 
 #############################################
	@echo off
	title Servidor Kafka 4.3.0 (KRaft Standalone)
	echo ====================================================
	echo    Iniciando Apache Kafka v4.3.0 en Modo KRaft
	echo ====================================================
	
	:: Forzar a trabajar desde la ruta raíz de Kafka donde está este script
	cd /d "%~dp0"
	
	:: Usamos la ruta absoluta del directorio de trabajo actual (%CD%)
	set "DATA_DIR=%CD%\data\kraft-combined-logs"
	set "CONFIG_FILE=%CD%\config\server.properties"
	
	echo [DEBUG] Verificando ruta de config: %CONFIG_FILE%
	
	:: Si no existe la carpeta data, se fuerza el formateo moderno
	if not exist "%DATA_DIR%" (
	    echo [INFO] Detectado entorno limpio o corrupto. Formateando almacenamiento...
	    
	    :: CORRECCIÓN: Añadido el flag --standalone requerido en Kafka 4.x
	    call "%CD%\bin\windows\kafka-storage.bat" format --standalone -t auwpwmsyRI63YSy8SbbIUA -c "%CONFIG_FILE%"
	    
	    echo ----------------------------------------------------
	)
	
	echo [INFO] Lanzando el Broker de Kafka...
	call "%CD%\bin\windows\kafka-server-start.bat" "%CONFIG_FILE%"
